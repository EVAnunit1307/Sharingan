cmake_minimum_required(VERSION 3.20)
project(wallhack_orb LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release)
if(NOT ORB_SOURCE OR NOT EIGEN_SOURCE)
  message(FATAL_ERROR "Set ORB_SOURCE and EIGEN_SOURCE to the pinned source directories")
endif()
find_package(OpenCV 4.4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d features2d highgui)
find_package(Boost CONFIG REQUIRED COMPONENTS serialization)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_compile_definitions(COMPILEDWITHC11)
include_directories(${ORB_SOURCE} ${ORB_SOURCE}/include ${ORB_SOURCE}/include/CameraModels
  ${ORB_SOURCE}/Thirdparty/Sophus ${EIGEN_SOURCE} ${OpenCV_INCLUDE_DIRS}
  ${CMAKE_CURRENT_SOURCE_DIR}/headless)
file(GLOB DBOW_SOURCES ${ORB_SOURCE}/Thirdparty/DBoW2/DBoW2/*.cpp ${ORB_SOURCE}/Thirdparty/DBoW2/DUtils/*.cpp)
add_library(dbow STATIC ${DBOW_SOURCES})
target_link_libraries(dbow PUBLIC ${OpenCV_LIBS} Boost::serialization)
file(READ ${ORB_SOURCE}/Thirdparty/g2o/CMakeLists.txt G2O_BUILD)
string(REGEX MATCHALL "g2o/[A-Za-z0-9_/]+\\.(cpp|c)" G2O_FILES "${G2O_BUILD}")
set(G2O_SOURCES)
foreach(source IN LISTS G2O_FILES)
  list(APPEND G2O_SOURCES ${ORB_SOURCE}/Thirdparty/g2o/${source})
endforeach()
add_library(g2o STATIC ${G2O_SOURCES})
file(GLOB ORB_SOURCES ${ORB_SOURCE}/src/*.cc ${ORB_SOURCE}/src/*.cpp ${ORB_SOURCE}/src/CameraModels/*.cpp)
list(FILTER ORB_SOURCES EXCLUDE REGEX "/(Viewer|MapDrawer)\\.cc$")
add_library(orb STATIC ${ORB_SOURCES} adapter.cc)
target_link_libraries(orb PUBLIC dbow g2o Boost::serialization OpenSSL::Crypto Threads::Threads ${OpenCV_LIBS})
add_executable(orb_replay replay.cc)
target_link_libraries(orb_replay PRIVATE orb)
